2 Afficher par mode de paiement le montant total des achats effectués
SELECT 
    payment as Mode_paiement,
    sum(cost) as total_paiement
FROM purchases
GROUP BY payment
ORDER BY total_paiement;



3 Affichez en une seule requête les taux de paiement par cash et par paiement
électronique
SELECT
    CASE
        WHEN lower(payment) = 'cash' THEN 'Cash'
        ELSE 'Electronique'
    END AS type_paiement,
    COUNT(*) AS total,
    COUNT(*) * 100.0 / SUM(COUNT(*)) OVER() AS pourcentage
FROM purchases
GROUP BY
    CASE
        WHEN lower(payment) = 'cash' THEN 'Cash'
        ELSE 'Electronique'
    END;
